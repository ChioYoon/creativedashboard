> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `step2_clustering.html`에 "N주/개월 전과 비교" 기능을 추가해, 두 시점의 군집화 결과를 Jaccard 기반으로 매칭하고 어떤 군집이 성장/하락했는지, 어떤 군집이 신규/소멸했는지 보여준다.

**Architecture:** git 히스토리(nightly 커밋)를 암묵적 스냅샷 저장소로 활용 — GitHub 커밋 API로 과거 시점의 `public/data/{title}.json` 커밋을 찾고 `raw.githubusercontent.com`에서 그 시점 JSON을 가져온다. 기존 군집화 알고리즘(`aggregateByMaterial`/`calcRankScores`/`buildTagClusters`/`assignMaterialsToClusters`/`mergeSmallClusters`/`enrichClusters`)은 이미 순수 함수로 분리돼 있으므로, `runClustering()`에서 전역 상태·DOM에 결합된 오케스트레이션 부분만 걷어내 재사용 가능한 `computeClusterResults(rows, cfg, onPhase)`로 추출한다. 과거 데이터도 이 함수로 동일하게 계산한 뒤, 소재 단위 Jaccard 유사도로 군집을 매칭해 비교 결과를 렌더링한다.

**Tech Stack:** 순수 프론트엔드(빌드 스텝 없는 정적 사이트). 신규 백엔드·저장소 없음 — GitHub REST API(공개 저장소, 무인증) + `raw.githubusercontent.com`. 이 프로젝트엔 JS용 자동화 테스트가 없으므로 검증은 `mcp__Claude_Preview__*` 도구로 브라우저에서 직접 확인한다.

`step2_clustering.html`의 다음 블록(현재 1589~1746줄, `/* ── 메인 실행 함수 ── */` 주석부터 `runClustering` 닫는 `}`까지):
```js
/* ── 메인 실행 함수 ── */
async function runClustering() {
  if (!csvData || !sessionConfig) {
    showToast('데이터가 없습니다.', 'error'); return;
  }

  const overlay = document.getElementById('loadingOverlay');
  overlay.classList.add('show');
  setLoadingPhase(0);
  updateLoadingMsg('소재명 기준으로 일자별 데이터 집계 중...');

  await sleep(100);

  try {
    const cfg        = sessionConfig;
    // ★ 마케터 인사이트는 군집화 태그에서 제외 (소재 설명용으로만 활용)
    const tagCols    = (cfg.columnConfig || []).filter(c => c.type === 'tag');
    // 인사이트 컬럼은 매핑 테이블 표시용으로 별도 보존
    const insightCols = (cfg.columnConfig || []).filter(c => c.type === 'insight');
    const params     = cfg.params || {};
    const perfWeight = cfg.perfWeight || 30;

    // ━━━ Phase 0: 소재명 기준 집계 ━━━
    // TXT 유형 제외: Step1 소재 성과 분석과 동일하게 BNR/VID만 분석 대상
    const bnrVidRows = csvData.rows.filter(row => {
      const t = (row['유형'] || row['앱 확장 소재 유형'] || '').toUpperCase().trim();
      return t === 'BNR' || t === 'VID';
    });
    const txtCount = csvData.rows.length - bnrVidRows.length;
    if (txtCount > 0) {
      console.log(`[군집화] TXT 제외: 전체 ${csvData.rows.length}개 → BNR/VID ${bnrVidRows.length}개 (TXT ${txtCount}개 제외)`);
    }

    // 일자별 raw rows → 소재명 단위로 전환/비용/노출/클릭/매출 합산
    updateLoadingMsg(`일자별 데이터를 소재명 기준으로 합산 중... (BNR/VID ${bnrVidRows.length}개${txtCount > 0 ? `, TXT ${txtCount}개 제외` : ''})`);
    const aggregated = aggregateByMaterial(bnrVidRows);
    const rawRowCount = bnrVidRows.length;
    const matCount    = aggregated.length;

    // ━━━ Rank 기반 스코어링 (Step1 동일) ━━━
    const scoreWeights = {
      conv: 0.25, cpa: 0.25, ipm: 0.25, roas: 0.25   // 기본 균등 가중치
    };
    const scoredMaterials = calcRankScores(aggregated, scoreWeights);

    // 집계 결과를 설정 요약에 반영
    updateAggregationBadge(rawRowCount, matCount);

    await sleep(200);

    // ━━━ Phase 1: 태그 빈도 & IQR 노이즈 필터 ━━━
    // 집계된 소재(_firstRow)의 태그 기반으로 빈도/공출현 계산
    setLoadingPhase(1);  // Phase 0 완료 → Phase 1 활성
    updateLoadingMsg(`IQR 기반 노이즈 태그 필터링 중... (${matCount}개 소재 대상)`);

    const tagFreqs = {};
    const weightMap = {};
    const coMatrix  = {};

    // ★ 포맷 자동 태그화: 파일명 패턴으로 사이즈/포맷 태그 부여
    function detectFormatTag(mat) {
      const src = (mat.파일명 || mat._key || mat.소재명 || '').toLowerCase()
                + (mat._firstRow?.['앱 확장 소재'] || '').toLowerCase();
      if (/_v_|1200x1500/i.test(src)) return 'format:vertical';
      if (/_s_|1200x1200/i.test(src)) return 'format:square';
      if (/_l_|1200x628/i.test(src))  return 'format:landscape';
      if (/vid|\.mp4/i.test(src))     return 'format:video';
      return null;
    }

    scoredMaterials.forEach(mat => {
      const srcRow = mat._firstRow;
      const rowTags = [];

      // ★ 포맷 태그 자동 추가
      const fmtTag = detectFormatTag(mat);
      if (fmtTag) {
        tagFreqs[fmtTag]  = (tagFreqs[fmtTag]  || 0) + 1;
        weightMap[fmtTag] = weightMap[fmtTag] || 50;
        rowTags.push(fmtTag);
      }

      tagCols.forEach(col => {
        const val = (srcRow[col.col] || '').trim();
        if (!val || val === '-') return;
        // ★ 태그 정규화(동의어 통합) 후 분리
        splitTagValue(val).map(v => normalizeTag(v)).filter(v => v && v.length < 60).forEach(v => {
          tagFreqs[v]  = (tagFreqs[v]  || 0) + 1;
          weightMap[v] = Math.max(weightMap[v] || 0, col.weight || 40);
          rowTags.push(v);
        });
      });
      // 공출현 매트릭스 (소재 단위로 계산)
      for (let i = 0; i < rowTags.length; i++) {
        for (let j = i + 1; j < rowTags.length; j++) {
          const key = [rowTags[i], rowTags[j]].sort().join('|||');
          coMatrix[key] = (coMatrix[key] || 0) + 1;
        }
      }
    });

    const noiseSet = iqrNoiseFilter(tagFreqs, scoredMaterials);
    await sleep(200);

    // ━━━ Phase 2: Union-Find 클러스터링 ━━━
    setLoadingPhase(2);
    updateLoadingMsg('태그 공출현 분석 및 군집 형성 중...');

    const n = scoredMaterials.length;
    // ★ co-occurrence 임계값: 고유 태그 수 기반 자동 결정
    //   태그가 적을수록 관대하게(넓게 연결) → 군집 형성 용이
    const uniqueTagCount = Object.keys(tagFreqs).length;
    const coThreshold = uniqueTagCount <= 10 ? 0.10
                      : uniqueTagCount <= 20 ? 0.15
                      : uniqueTagCount <= 40 ? 0.20
                      : 0.25;
    // ★ 최소 군집 수: 파라미터 설정값 사용 (기본 2개)
    const minClusters = Math.max(2, parseInt(params.minClusters) || 2);
    const tagGroups = buildTagClusters(tagFreqs, coMatrix, noiseSet, weightMap, coThreshold, minClusters);
    await sleep(300);

    // ━━━ Phase 3: 소재 배정 (집계+스코어링된 소재 사용) ━━━
    setLoadingPhase(3);
    updateLoadingMsg('소재를 군집에 배정하고 TotalScore 기반 정제 중...');

    const rawClusters = assignMaterialsToClusters(
      scoredMaterials, tagGroups, tagCols, weightMap, insightCols, perfWeight
    );
    // ⓐ 최소 군집 크기 적용 — 크기 미만 군집을 가장 유사한 큰 군집에 흡수 ('기타'는 유지)
    const minClusterSize = Math.max(2, parseInt(params.minCluster) || 3);
    const mergedClusters = mergeSmallClusters(rawClusters, minClusterSize);
    console.log(`[군집화] 최소 군집 크기 ${minClusterSize} 적용: ${rawClusters.length}개 → ${mergedClusters.length}개 군집`);
    const enriched = enrichClusters(mergedClusters, weightMap);
    await sleep(300);

    // ━━━ Phase 4: 인사이트 도출 ━━━
    setLoadingPhase(4);
    updateLoadingMsg('패턴 분석 및 인사이트 도출 중...');
    await sleep(200);

    clusterResults = {
      clusters: enriched,
      tagFreqs,
      weightMap,
      noiseSet: [...noiseSet],
      params: { coThreshold, perfWeight, n, rawRowCount, matCount }
    };

    overlay.classList.remove('show');
    renderResults(clusterResults);
    showToast(`✅ 군집화 완료! ${matCount}개 소재 → ${enriched.length}개 군집`, 'success');

  } catch (err) {
    overlay.classList.remove('show');
    console.error(err);
    showToast('군집화 오류: ' + err.message, 'error');
  }
}
```

```js
   과거 스냅샷 계산은 생략 가능(느슨한 pacing sleep은 이 함수가 그대로 소유). */

    if (/vid|\.mp4/i.test(src))     return 'format:video';

  // ★ co-occurrence 임계값: 고유 태그 수 기반 자동 결정
  //   태그가 적을수록 관대하게(넓게 연결) → 군집 형성 용이

  const coThreshold = uniqueTagCount <= 10 ? 0.10
```

```js
  const rawUrl = `https://raw.githubusercontent.com/${GITHUB_REPO}/${sha}/${path}`;
  const rawRes = await fetch(rawUrl);
  if (!rawRes.ok) throw new Error(`과거 데이터 조회 실패 (HTTP ${rawRes.status})`);

  const normalized = window.DataSource.normalizeFromJson(historicalJson);

  catch (e) { /* sessionStorage 용량 초과 등 — 캐시는 최적화일 뿐이므로 무시 */ }
```

## Task 3: 군집 매칭 알고리즘 (`matchClustersAcrossSnapshots`)

두 시점의 군집을 소재 단위 Jaccard 유사도로 매칭한다.

**Files:**
- Modify: `step2_clustering.html` (Task 2의 `fetchHistoricalSnapshot` 뒤에 신규 함수 추가)

**Interfaces:**
- Consumes: 군집 객체 `{materials: [{_key, 소재명, 파일명, ...}], avgScore, grade, name, ...}` (Task 1 `computeClusterResults` 및 기존 `enrichClusters`가 만드는 형태와 동일).
- Produces: `function matchClustersAcrossSnapshots(nowClusters, prevClusters, threshold)` → `{matched: [{now, prev, jaccard}], goneClusters: [cluster,...]}`. `matched` 배열의 각 항목은 `prev`가 `null`이면 "신규 군집"(과거에 대응 없음), 아니면 매칭된 과거 군집.

- [ ] **Step 1: `fetchHistoricalSnapshot` 뒤에 다음 코드 추가**

```js
/* ── 군집 트렌드 비교 — Jaccard 기반 군집 매칭 ──
   Union-Find 재계산 시 군집 ID는 시점마다 안정적이지 않으므로, 소재 구성 겹침으로 매칭한다.
   threshold 미만이면 매칭 안 함(신규 취급) — 실측 데이터 부족한 초기 추정치, 조정 여지 있음. */
function _clusterMaterialKeys(cluster) {
  return new Set((cluster.materials || []).map(m => m._key || m.소재명 || m.파일명 || ''));
}

function _jaccardSets(a, b) {
  if (!a.size || !b.size) return 0;
  let inter = 0;
  a.forEach(k => { if (b.has(k)) inter++; });
  const union = a.size + b.size - inter;
  return union ? inter / union : 0;
}

function matchClustersAcrossSnapshots(nowClusters, prevClusters, threshold) {
  threshold = threshold == null ? 0.2 : threshold;
  const prevKeySets = prevClusters.map(c => _clusterMaterialKeys(c));

  const matched = [];
  const matchedPrevIdx = new Set();

  nowClusters.forEach(nowC => {
    const nowKeys = _clusterMaterialKeys(nowC);
    let bestIdx = -1, bestJ = 0;
    prevClusters.forEach((prevC, i) => {
      const j = _jaccardSets(nowKeys, prevKeySets[i]);
      if (j > bestJ) { bestJ = j; bestIdx = i; }
    });
    if (bestIdx >= 0 && bestJ >= threshold) {
      matched.push({ now: nowC, prev: prevClusters[bestIdx], jaccard: bestJ });
      matchedPrevIdx.add(bestIdx);
    } else {
      matched.push({ now: nowC, prev: null, jaccard: 0 });   // 과거 대응 없음 = 신규 군집
    }
  });

  const goneClusters = prevClusters.filter((c, i) => !matchedPrevIdx.has(i));

  return { matched, goneClusters };
}
```

```js
    <div class="ic-header"><span class="ic-icon">📈</span><span class="ic-title">상승/하락 군집</span></div>
    <div class="ic-body" style="display:flex;gap:24px;flex-wrap:wrap;">
      <div style="flex:1;min-width:220px;">
        <div style="font-size:12px;font-weight:700;color:var(--mid);margin-bottom:6px;">🔼 상승</div>
        ${risers.length ? risers.map(deltaRow).join('') : '<div style="font-size:11px;color:var(--gray-text);padding:6px 0;">뚜렷한 상승 없음</div>'}
      </div>
      <div style="flex:1;min-width:220px;">
        <div style="font-size:12px;font-weight:700;color:var(--mid);margin-bottom:6px;">🔽 하락</div>
        ${fallers.length ? fallers.map(deltaRow).join('') : '<div style="font-size:11px;color:var(--gray-text);padding:6px 0;">뚜렷한 하락 없음</div>'}
      </div>
    </div>
  </div>`;

  const listNames = (clusters) => clusters.length ? clusters.map(c => escHtml(c.name)).join(', ') : '없음';

  const newGoneCard = `<div class="insight-card fade-in" style="grid-column:1/-1;">
    <div class="ic-header"><span class="ic-icon">🆕</span><span class="ic-title">신규 · 소멸 군집</span></div>
    <div class="ic-body">
      <div style="margin-bottom:8px;"><strong>신규:</strong> ${listNames(newOnes)}</div>
      <div><strong>소멸:</strong> ${listNames(goneOnes)}</div>
    </div>
  </div>`;

  return `<div class="insight-grid">${summaryCard}${newGoneCard}</div>`;
}
```
