다음 함수 전체(현재 4965~4993)를 삭제(뒤따르는 빈 줄 1개 포함, 다음 주석 `// 위험 신호 자동 감지`는 유지):
```js
    function buildSparklineSvg(values, opts = {}) {
      const width = opts.width || 80;
      const height = opts.height || 22;
      const stroke = opts.stroke || '#DC2828';
      const fill = opts.fill || 'rgba(220,40,40,0.15)';
      if (!values || values.length < 2) {
        return `<span style="font-size:10px;color:#d1d5db;">—</span>`;
      }
      const max = Math.max(...values, 0);
      const min = Math.min(...values);
      const range = max - min || 1;
      const step = width / (values.length - 1);
      const points = values.map((v, i) => {
        const x = (i * step).toFixed(1);
        const y = (height - 2 - ((v - min) / range) * (height - 4)).toFixed(1);
        return `${x},${y}`;
      });
      const polyPoints = points.join(' ');
      const areaPath = `M0,${height} L${points.join(' L')} L${width},${height} Z`;
      // 마지막 점 표시
      const lastIdx = values.length - 1;
      const lastX = (lastIdx * step).toFixed(1);
      const lastY = (height - 2 - ((values[lastIdx] - min) / range) * (height - 4)).toFixed(1);
      return `<svg width="${width}" height="${height}" style="overflow:visible;display:block;">
        <path d="${areaPath}" fill="${fill}" stroke="none"/>
        <polyline points="${polyPoints}" fill="none" stroke="${stroke}" stroke-width="1.5"/>
        <circle cx="${lastX}" cy="${lastY}" r="2" fill="${stroke}"/>
      </svg>`;
    }
```

- [ ] **Step 8: `buildDailySparklineMap` 정의 삭제**

다음 함수 전체(현재 4934~4963)를 삭제(앞뒤 다른 함수는 유지):
```js
    function buildDailySparklineMap(rows, groupBy) {
      const dateColumn = detectDateColumn(normalizedData.columns);
      if (!dateColumn) return {};
      const map = {}; // {key: [{date, 전환, CPA}, ...]}
      rows.forEach(r => {
        const key = r[groupBy];
        if (!key) return;
        const date = r[dateColumn];
        if (!date) return;
        if (!map[key]) map[key] = {};
        if (!map[key][date]) map[key][date] = {전환:0, 비용:0, 노출수:0};
        map[key][date].전환 += safeParseNumber(r['전환']);
        map[key][date].비용 += safeParseNumber(r['비용']);
        map[key][date].노출수 += safeParseNumber(r['노출수']);
      });
      // 정렬된 배열로 변환
      const result = {};
      Object.keys(map).forEach(key => {
        const dailyArr = Object.entries(map[key])
          .map(([date, v]) => ({
            date,
            전환: v.전환,
            CPA: v.전환 > 0 ? v.비용 / v.전환 : 0,
            노출수: v.노출수
          }))
          .sort((a, b) => a.date.localeCompare(b.date));
        result[key] = dailyArr;
      });
      return result;
    }
```
