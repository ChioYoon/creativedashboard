- [ ] **Step 3: renderModalTrend 함수**

`openPreviewModal` 함수 닫힘 `}`(6840) 다음에 추가:
```js
    function renderModalTrend(creative) {
      const cv = document.getElementById('modalTrendChart');
      if (!cv || typeof Chart === 'undefined') return;
      const daily = ((creative.meta && creative.meta.kpi_daily) || []).slice()
        .sort((a, b) => String(a.date).localeCompare(String(b.date)));
      const labels = daily.map(d => String(d.date).slice(5));
      const series = {
        '전환': daily.map(d => Number(d.conversions || 0)),
        '노출': daily.map(d => Number(d.impressions || 0)),
        'CPA':  daily.map(d => (Number(d.conversions) > 0 ? Math.round(Number(d.cost || 0) / Number(d.conversions)) : null)),
      };
      const btns = document.querySelectorAll('.trend-metric-btn');
      const setActive = (m) => btns.forEach(b => { const on = b.dataset.metric === m; b.style.background = on ? '#E84855' : '#fff'; b.style.color = on ? '#fff' : '#374151'; });
      const draw = (metric) => {
        if (window._modalTrendChart) { try { window._modalTrendChart.destroy(); } catch (_) {} }
        window._modalTrendChart = new Chart(cv.getContext('2d'), {
          type: 'line',
          data: { labels, datasets: [{ label: metric, data: series[metric], borderColor: '#E84855', backgroundColor: 'rgba(232,72,85,.08)', tension: .25, spanGaps: true, pointRadius: 2, fill: true }] },
          options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true } } }
        });
        setActive(metric);
      };
      btns.forEach(b => b.addEventListener('click', () => draw(b.dataset.metric)));
      draw('전환');
    }
```
